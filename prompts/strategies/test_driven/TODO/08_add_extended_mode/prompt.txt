### Context
`Task` currently has no project concept. You need to introduce a `Project` domain class and extend `Task` with an optional `project_id` so tasks can be grouped and filtered by project. Existing stored tasks without `project_id` must remain loadable. Projects must also be manageable via CLI commands and the interactive menu (`python -m src`).

### Objective
Implement `Project` and `ProjectService`, add optional `project_id` to `Task`, extend `list_tasks` to support filtering by project, and expose project management and task-project assignment in the CLI and interactive menu.

### Test Suite
```python
import pytest
from datetime import datetime, timezone
from src.models.project import Project
from src.services.todo_service import TodoService
from src.services.project_service import ProjectService
from src.storage.json_storage import JsonStorage
from src.cli.todo_cli import TodoCLI


@pytest.fixture
def services(tmp_path):
    storage = JsonStorage(str(tmp_path / "tasks.json"))
    todo = TodoService(storage)
    return todo, ProjectService(todo)


def test_project_can_be_created():
    assert Project(name="Work") is not None


def test_project_has_unique_id():
    assert Project(name="Work").id != Project(name="Work").id


def test_empty_project_name_raises():
    with pytest.raises(Exception):
        Project(name="")


def test_create_and_list_projects(services):
    _, projects = services
    p = projects.create("Work")
    assert any(x.id == p.id for x in projects.list())


def test_task_assigned_to_project(services):
    todo, projects = services
    p = projects.create("Work")
    task = todo.add_task("Task", project_id=p.id)
    assert task.project_id == p.id


def test_list_tasks_by_project(services):
    todo, projects = services
    p = projects.create("Work")
    todo.add_task("Task A", project_id=p.id)
    todo.add_task("Task B")
    assert all(t.project_id == p.id for t in todo.list_tasks(project_id=p.id))


def test_task_without_project_id_is_none(services):
    todo, _ = services
    assert todo.add_task("No project").project_id is None

def test_project_id_is_uuid_string():
    import uuid
    project = Project(name="Work")
    parsed = uuid.UUID(project.id)
    assert str(parsed) == project.id

def test_old_tasks_without_project_id_load_fine(tmp_path):
    storage = JsonStorage(str(tmp_path / "tasks.json"))
    storage.save([{
        "id": "abc", "title": "Old", "description": None, "status": "pending",
        "created_at": datetime.now(timezone.utc).isoformat(),
        "updated_at": datetime.now(timezone.utc).isoformat(),
    }])
    assert TodoService(storage).list_tasks()[0].project_id is None


def test_move_task_between_projects(services):
    todo, projects = services
    p1 = projects.create("Alpha")
    p2 = projects.create("Beta")
    task = todo.add_task("Movable", project_id=p1.id)
    assert todo.update_task(task.id, project_id=p2.id).project_id == p2.id


def test_cli_project_create(tmp_path, capsys):
    cli = TodoCLI(str(tmp_path / "tasks.json"))
    rc = cli.run(["project", "create", "Work"])
    assert rc == 0
    assert "Work" in capsys.readouterr().out


def test_cli_project_list(tmp_path, capsys):
    storage_path = str(tmp_path / "tasks.json")
    cli = TodoCLI(storage_path)
    cli.run(["project", "create", "Work"])
    capsys.readouterr()
    cli.run(["project", "list"])
    assert "Work" in capsys.readouterr().out


def test_cli_add_task_with_project(tmp_path):
    storage_path = str(tmp_path / "tasks.json")
    todo = TodoService(JsonStorage(storage_path))
    proj = ProjectService(todo).create("Work")
    cli = TodoCLI(storage_path)
    rc = cli.run(["add", "My task", "--project", proj.id])
    assert rc == 0
    tasks = TodoService(JsonStorage(storage_path)).list_tasks()
    assert any(t.project_id == proj.id for t in tasks)


def test_cli_list_filter_by_project(tmp_path, capsys):
    storage_path = str(tmp_path / "tasks.json")
    todo = TodoService(JsonStorage(storage_path))
    proj = ProjectService(todo).create("Work")
    todo.add_task("Work task", project_id=proj.id)
    todo.add_task("Personal task")
    cli = TodoCLI(storage_path)
    cli.run(["list", "--project", proj.id])
    out = capsys.readouterr().out
    assert "Work task" in out
    assert "Personal task" not in out
```

### Implementation Rules
- Do not modify the tests.
- Tasks without `project_id` must continue to work exactly as before.
- Existing stored JSON without `project_id` must load without error.
- Do not implement drag-and-drop UI or per-project access control.
- `Project.id` must be a UUID string generated automatically.
- `project_id` must be optional on `Task` and default to `None`.
- Project names must be stripped or validated so blank/whitespace-only names are rejected.
- Project filtering must not change existing `list_tasks()` behavior when no `project_id` filter is provided.
- Add project management to the interactive menu: create/list projects, assign a project when adding a task, filter the task list by project.

- All provided tests pass.
- Existing tests still pass.
- Code compiles without syntax or import errors.
- `Task.from_dict` handles records missing `project_id` without raising.
